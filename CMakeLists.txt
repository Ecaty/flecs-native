cmake_minimum_required(VERSION 3.20)
project(FlecsUnityCore LANGUAGES C CXX)
# Third-party code lives in vendor/ as submodules of our forks; the submodule commit is the pin.
set(FLECS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/flecs")
if(NOT EXISTS "${FLECS_DIR}/distr/flecs.c")
  message(FATAL_ERROR "Missing vendor submodules. Run: git submodule update --init --recursive")
endif()
# The binary reports FLECS_VERSION+<short pinned commit>. Read it from the submodule checkout;
# fall back to the gitlink recorded in this repository, then to "unknown".
find_package(Git QUIET)
set(FLECS_UPSTREAM_COMMIT "")
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${FLECS_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE FLECS_UPSTREAM_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE FLECS_GIT_RESULT)
  if(NOT FLECS_GIT_RESULT EQUAL 0 OR NOT FLECS_UPSTREAM_COMMIT MATCHES "^[0-9a-f]+$")
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse HEAD:vendor/flecs
      OUTPUT_VARIABLE FLECS_UPSTREAM_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
endif()
if(FLECS_UPSTREAM_COMMIT MATCHES "^[0-9a-f]+$")
  string(SUBSTRING "${FLECS_UPSTREAM_COMMIT}" 0 7 FLECS_UPSTREAM_SHORT)
else()
  set(FLECS_UPSTREAM_SHORT "unknown")
endif()
message(STATUS "Flecs submodule revision: ${FLECS_UPSTREAM_SHORT}")
# Our own sources (vendor code is compiled as it ships).
set(FU_CORE_SOURCES src/flecs_unity.cpp src/Components.cpp src/TypeGuard.cpp src/Views.cpp
  src/Capabilities.cpp src/Geometry.cpp src/BehaviorTrees.cpp src/Audio.cpp)
add_library(flecs_unity_core OBJECT src/flecs_unity.cpp src/Components.cpp src/TypeGuard.cpp "${FLECS_DIR}/distr/flecs.c")
# -DFU_WARNINGS=ON compiles our sources with warnings. Warnings that vendor headers raise inside these files are
# expected; a warning located in src/ is ours to fix (the CI step checks for those).
option(FU_WARNINGS "Compile the core's own sources with compiler warnings" OFF)
if(FU_WARNINGS)
  if(MSVC)
    set_source_files_properties(${FU_CORE_SOURCES} PROPERTIES COMPILE_OPTIONS "/W4")
  else()
    set_source_files_properties(${FU_CORE_SOURCES} PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra")
  endif()
endif()
# Keep only flecs.h on the include path (copied), never the Flecs repository root or vendor/.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/flecs_include")
configure_file("${FLECS_DIR}/distr/flecs.h" "${CMAKE_CURRENT_BINARY_DIR}/flecs_include/flecs.h" COPYONLY)
target_include_directories(flecs_unity_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/flecs_include")
target_compile_features(flecs_unity_core PUBLIC c_std_99 cxx_std_17)
target_compile_definitions(flecs_unity_core PUBLIC FLECS_SOFT_ASSERT FLECS_KEEP_ASSERT FLECS_CREATE_MEMBER_ENTITIES FLECS_SCRIPT_MATH FU_FLECS_REVISION="${FLECS_UPSTREAM_SHORT}")
set_target_properties(flecs_unity_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_property(TARGET flecs_unity_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(flecs_unity_core PRIVATE /utf-8)
endif()
if(WIN32)
  target_link_libraries(flecs_unity_core PUBLIC ws2_32 dbghelp)
elseif(ANDROID)
  # Bionic has pthread in libc; log for __android_log_print.
  target_link_libraries(flecs_unity_core PUBLIC m log)
elseif(UNIX)
  target_link_libraries(flecs_unity_core PUBLIC m pthread)
endif()

# Portable capabilities are linked once into the same plugin and Flecs world.
set(BTCPP_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BTCPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BTCPP_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BTCPP_GROOT_INTERFACE OFF CACHE BOOL "" FORCE)
set(BTCPP_SQLITE_LOGGING OFF CACHE BOOL "" FORCE)
set(CLIPPER2_UTILS OFF CACHE BOOL "" FORCE)
set(CLIPPER2_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CLIPPER2_TESTS OFF CACHE BOOL "" FORCE)
set(CLIPPER2_USINGZ OFF CACHE STRING "" FORCE)
add_subdirectory(vendor/behaviortree EXCLUDE_FROM_ALL)
add_subdirectory(vendor/clipper2/CPP EXCLUDE_FROM_ALL)
target_sources(flecs_unity_core PRIVATE src/Views.cpp src/Capabilities.cpp src/Geometry.cpp src/BehaviorTrees.cpp src/Audio.cpp)
target_include_directories(flecs_unity_core PRIVATE vendor/tinysoundfont)
target_link_libraries(flecs_unity_core PUBLIC behaviortree_cpp Clipper2)

# -DFU_BUILD_TESTS=ON builds flecs_unity_core_tests: checks for logic the C ABI smokes cannot reach directly.
option(FU_BUILD_TESTS "Build the core test executable" OFF)
if(FU_BUILD_TESTS)
  add_executable(flecs_unity_core_tests tests/core_tests.cpp)
  target_link_libraries(flecs_unity_core_tests PRIVATE flecs_unity_core)
  if(MINGW)
    target_link_options(flecs_unity_core_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
