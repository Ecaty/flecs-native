cmake_minimum_required(VERSION 3.20)
project(EcatyFlecsNative LANGUAGES C CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/UPSTREAM.json" FLECS_UPSTREAM_JSON)
string(JSON FLECS_UPSTREAM_COMMIT GET "${FLECS_UPSTREAM_JSON}" commit)
string(SUBSTRING "${FLECS_UPSTREAM_COMMIT}" 0 7 FLECS_UPSTREAM_SHORT)
add_library(ecaty_flecs_core OBJECT src/flecs_unity.cpp src/Components.cpp vendor/flecs.c)
# Do not put vendor/ on the standard include path: its VERSION file shadows <version> on Windows.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/flecs_include")
configure_file(vendor/flecs.h "${CMAKE_CURRENT_BINARY_DIR}/flecs_include/flecs.h" COPYONLY)
target_include_directories(ecaty_flecs_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/flecs_include")
target_compile_features(ecaty_flecs_core PUBLIC c_std_99 cxx_std_17)
target_compile_definitions(ecaty_flecs_core PUBLIC FLECS_SOFT_ASSERT FLECS_KEEP_ASSERT FLECS_CREATE_MEMBER_ENTITIES FLECS_SCRIPT_MATH FU_FLECS_REVISION="${FLECS_UPSTREAM_SHORT}")
set_target_properties(ecaty_flecs_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_property(TARGET ecaty_flecs_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ecaty_flecs_core PRIVATE /utf-8)
endif()
if(WIN32)
  target_link_libraries(ecaty_flecs_core PUBLIC ws2_32 dbghelp)
elseif(UNIX)
  target_link_libraries(ecaty_flecs_core PUBLIC m pthread)
endif()

# Portable capabilities are linked once into the same plugin and Flecs world.
set(BTCPP_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BTCPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BTCPP_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BTCPP_GROOT_INTERFACE OFF CACHE BOOL "" FORCE)
set(BTCPP_SQLITE_LOGGING OFF CACHE BOOL "" FORCE)
set(CLIPPER2_UTILS OFF CACHE BOOL "" FORCE)
set(CLIPPER2_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CLIPPER2_TESTS OFF CACHE BOOL "" FORCE)
set(CLIPPER2_USINGZ OFF CACHE STRING "" FORCE)
add_subdirectory(vendor/behaviortree EXCLUDE_FROM_ALL)
add_subdirectory(vendor/clipper2/CPP EXCLUDE_FROM_ALL)
target_sources(ecaty_flecs_core PRIVATE src/RuntimeCapabilities.cpp)
target_include_directories(ecaty_flecs_core PRIVATE vendor/tinysoundfont)
target_link_libraries(ecaty_flecs_core PUBLIC behaviortree_cpp Clipper2)
